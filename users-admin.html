let allUsers = []; function escapeHtml(value) { return String(value || "") .replaceAll("&", "&") .replaceAll("<", "<") .replaceAll(">", ">") .replaceAll('"', """) .replaceAll("'", "'"); } function formatDate(dateValue) { if (!dateValue) { return "Date inconnue"; } const date = new Date(dateValue); if (Number.isNaN(date.getTime())) { return "Date inconnue"; } return date.toLocaleString( "fr-FR", { day: "2-digit", month: "2-digit", year: "numeric", hour: "2-digit", minute: "2-digit" } ); } function renderUsers(users) { const container = document.getElementById( "usersContainer" ); const summary = document.getElementById( "usersSummary" ); summary.textContent = `${users.length} utilisateur(s) affiché(s)`; if (!users.length) { container.innerHTML = `
Aucun utilisateur trouvé.
`; return; } container.innerHTML = users.map((user) => { const email = escapeHtml(user.email); const whatsapp = escapeHtml( user.whatsapp || "Non renseigné" ); const userId = escapeHtml(user.id); const createdAt = formatDate(user.created_at); const isLeader = Boolean(user.is_leader); return `
${email}
📱 WhatsApp : ${whatsapp}
🆔 Identifiant : ${userId}
📅 Inscrit le : ${createdAt}
${isLeader ? "Leader" : "Membre"}
`; }).join(""); } function filterUsers() { const searchValue = document.getElementById( "searchInput" ) .value .trim() .toLowerCase(); if (!searchValue) { renderUsers(allUsers); return; } const filteredUsers = allUsers.filter((user) => { const email = String(user.email || "") .toLowerCase(); const whatsapp = String(user.whatsapp || "") .toLowerCase(); const id = String(user.id || "") .toLowerCase(); return ( email.includes(searchValue) || whatsapp.includes(searchValue) || id.includes(searchValue) ); }); renderUsers(filteredUsers); } async function loadUsers() { const container = document.getElementById( "usersContainer" ); try { const users = await adminApiCall( "/admin/users" ); allUsers = Array.isArray(users) ? users : []; renderUsers(allUsers); } catch (error) { container.innerHTML = `
${escapeHtml( error.message || "Impossible de charger les utilisateurs." )}
`; document.getElementById( "usersSummary" ).textContent = "Erreur de chargement"; } } document.getElementById( "searchInput" ).addEventListener( "input", filterUsers ); protectAdminPage(); loadUsers();
