// Dans login-admin.html async function loginAdmin() { const email = emailInput.value.trim().toLowerCase(); const password = passwordInput.value; if (!email || !password) { showAdminMessage(message, "Entre ton email et ton mot de passe."); return; } try { button.disabled = true; button.textContent = "Connexion..."; message.textContent = ""; // Appel API const response = await fetch(`${API_BASE_URL}/admin/login`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.message || data.error || 'Identifiants incorrects'); } // Sauvegarder le token if (data.token) { localStorage.setItem('point_focal_admin_token', data.token); } if (data.user) { localStorage.setItem('point_focal_admin_user', JSON.stringify(data.user)); } // Redirection window.location.href = 'dashboard-admin.html'; } catch (error) { showAdminMessage(message, error.message || "La connexion a échoué."); button.disabled = false; button.textContent = "Se connecter"; } }
